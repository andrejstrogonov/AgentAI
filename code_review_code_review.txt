```python
class ContextBuilder:
    def __init__(self, project_dir: str, config: dict):
        self.project_dir = Path(project_dir)
        self.config = config
        self.context_parts: list[str] = []
        self.max_tokens: int = config.get("max_context_tokens", 100_000)
        self._token_count: int = 0

    def add_file_contents(self, file_paths: list[Path]) -> "ContextBuilder":
        """Add file contents with token budget enforcement."""
        for path in file_paths:
            if self._token_count >= self.max_tokens:
                self.context_parts.append(
                    f"\n[TRUNCATED: token budget ({self.max_tokens}) reached]"
                )
                break
            try:
                content = path.read_text(encoding="utf-8", errors="replace")
                tokens_estimate = len(content) // 4  # rough approximation
                self._token_count += tokens_estimate
                self.context_parts.append(
                    f"\n--- START: {path.name} ---\n{content}\n--- END: {path.name} ---"
                )
            except OSError as e:
                self.context_parts.append(f"\n[ERROR reading {path.name}: {e}]")
        return self

    def add_dependencies(self) -> "ContextBuilder":
        """Parse and include dependency files."""
        dep_files = ["requirements.txt", "pyproject.toml", "setup.py", "Pipfile"]
        for dep_file in dep_files:
            dep_path = self.project_dir / dep_file
            if dep_path.exists():
                try:
                    content = dep_path.read_text(encoding="utf-8")
                    self.context_parts.append(
                        f"\n=== DEPENDENCIES ({dep_file}) ===\n{content}"
                    )
                except OSError as e:
                    self.context_parts.append(f"\n[ERROR reading {dep_file}: {e}]")
        return self

    def add_statistics(self) -> "ContextBuilder":
        """Add project statistics."""
        py_files = list(self.project_dir.rglob("*.py"))
        total_lines = 0
        for f in py_files:
            try:
                total_lines += sum(1 for _ in f.open(encoding="utf-8", errors="replace"))
            except OSError:
                pass
        self.context_parts.append(
            f"\n=== PROJECT STATS ===\n"
            f"Python files: {len(py_files)}\n"
            f"Total lines: {total_lines}\n"
            f"Estimated tokens used: {self._token_count}"
        )
        return self

    def build_prompt_context(self) -> str:
        """Finalize and return the assembled context string."""
        return "\n".join(self.context_parts)
```
